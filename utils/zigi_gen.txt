# Define the contents of the script with CLI argument parsing
script_content = """
import asyncio
import argparse
from benchmaker.sdk_api_wrapper.async_sdk_benchmaker_client import JsonKeyReaderClient

async def simulate_user(base_url, username, password):
    client = JsonKeyReaderClient(base_url, username, password)
    await client.login()
    await client.store("A", {"x": {"value": 5}, "y": {"value": 3}})
    await client.store("B", {"x": {"value": 2}, "y": {"value": 1}})
    await client.run_math("add", ["A", "B"])

async def main():
    parser = argparse.ArgumentParser(description="Run concurrent SDK load simulation.")
    parser.add_argument("--users", type=int, default=5, help="Number of concurrent users to simulate")
    parser.add_argument("--base-url", type=str, default="http://127.0.0.1:8010", help="Base URL of the API")
    parser.add_argument("--username", type=str, default="admin", help="Username for login")
    parser.add_argument("--password", type=str, default="admin123", help="Password for login")
    args = parser.parse_args()

    tasks = [
        simulate_user(args.base_url, args.username, args.password)
        for _ in range(args.users)
    ]
    await asyncio.gather(*tasks)

if __name__ == "__main__":
    asyncio.run(main())
"""

# Write the script to a file
script_path = Path("benchmaker/sdk_api_wrapper/async_load_runner.py")
script_path.write_text(script_content)

script_path.name  # Return just the filename for confirmation
